var (
	logLevel        = "INFO"
	debugLogs       = flag.Bool("debug", false, "Enable debug logs?")
	dbDialect       = flag.String("db-dialect", "sqlite3", "Database dialect (sqlite3 or postgres)")
	dbAddress       = flag.String("db-address", "file:whatsapp.db?_foreign_keys=on", "Database address")
	requestFullSync = flag.Bool("request-full-sync", false, "Request full (1 year) history sync when logging in?")
	outboxDir       = flag.String("outbox-dir", "outbox", "Outbox directory for outgoing messages")
	apiPort         = "" // loaded from API_PORT env var at startup, default :8080
	webhookURL      = "" // loaded from WEBHOOK env var at startup
	apiServer       *http.Server
	pairRejectChan  = make(chan bool, 1)
)

// clientInfo wraps a whatsmeow.Client with metadata for display.
type clientInfo struct {
	client *whatsmeow.Client
	phone  string // phone number for logging, e.g. "628123456789"
}

// clientManager manages multiple WhatsApp clients.
type clientManager struct {
	mu      sync.RWMutex
	clients map[string]*clientInfo // key = device JID string
}

func (cm *clientManager) add(jid string, ci *clientInfo) {
	cm.mu.Lock()
	cm.clients[jid] = ci
	cm.mu.Unlock()
}

func (cm *clientManager) remove(jid string) {
	cm.mu.Lock()
	delete(cm.clients, jid)
	cm.mu.Unlock()
}

func (cm *clientManager) get(jid string) *clientInfo {
	cm.mu.RLock()
	defer cm.mu.RUnlock()
	return cm.clients[jid]
}

func (cm *clientManager) list() []*clientInfo {
	cm.mu.RLock()
	defer cm.mu.RUnlock()
	list := make([]*clientInfo, 0, len(cm.clients))
	for _, ci := range cm.clients {
		list = append(list, ci)
	}
	return list
}

func (cm *clientManager) count() int {
	cm.mu.RLock()
	defer cm.mu.RUnlock()
	return len(cm.clients)
}

// findByPhone finds a client by phone number (e.g. "628123456789").
// Returns nil if not found.
func (cm *clientManager) findByPhone(phone string) *clientInfo {
	cm.mu.RLock()
	defer cm.mu.RUnlock()
	for _, ci := range cm.clients {
		if ci.phone == phone {
			return ci
		}
	}
	return nil
}

// random returns a random client, or nil if no clients are connected.
func (cm *clientManager) random() *clientInfo {
	list := cm.list()
	if len(list) == 0 {
		return nil
	}
	return list[rand.Intn(len(list))]
}

// hasPhone checks if a phone number is already connected.
func (cm *clientManager) hasPhone(phone string) bool {
	return cm.findByPhone(phone) != nil
}

// findKeyByPhone returns the JID key for a given phone number.
func (cm *clientManager) findKeyByPhone(phone string) string {
	cm.mu.RLock()
	defer cm.mu.RUnlock()
	for jid, ci := range cm.clients {
		if ci.phone == phone {
			return jid
		}
	}
	return ""
}

var clients = &clientManager{clients: make(map[string]*clientInfo)}

// storeContainer holds the database for all devices.
var storeContainer *sqlstore.Container

func ptr[T any](v T) *T {
	return &v
}

func main() {
	flag.Parse()

	// Load config from environment (via .env or system env).
	webhookURL = os.Getenv("WEBHOOK")
	if webhookURL == "" {
		log.Println("WARNING: WEBHOOK not set in .env — auto-reply will be disabled")
	}
	apiPort = os.Getenv("API_PORT")
	if apiPort == "" {
		apiPort = ":8080"
	} else if !strings.HasPrefix(apiPort, ":") {
		apiPort = ":" + apiPort
	}

	ctx := context.Background()
	if *debugLogs {
		logLevel = "DEBUG"
	}
	if *requestFullSync {
		store.DeviceProps.RequireFullSync = ptr(true)
		store.DeviceProps.HistorySyncConfig = &waProto.DeviceProps_HistorySyncConfig{
			FullSyncDaysLimit:   ptr(uint32(3650)),
			FullSyncSizeMbLimit: ptr(uint32(102400)),
			StorageQuotaMb:      ptr(uint32(102400)),
		}
	}

	dbLog := waLog.Stdout("Database", logLevel, true)
	var err error
	storeContainer, err = sqlstore.New(ctx, *dbDialect, *dbAddress, dbLog)
	if err != nil {
		log.Fatalf("Failed to connect to database: %v", err)
	}

	// Load all already-logged-in devices.
	devices, err := storeContainer.GetAllDevices(ctx)
	if err != nil {
		log.Fatalf("Failed to get devices: %v", err)
	}

	log.Printf("Found %d device(s) in database", len(devices))
	for _, device := range devices {
		if device.ID == nil {
			// Not yet paired — skip.
			continue
		}
		phone := extractPhone(device.ID)
		log.Printf("Connecting existing device: %s", phone)
		connectDevice(ctx, device, phone)
	}

	// Jika belum ada nomor, langsung tawarkan scan.
	if clients.count() == 0 {
		log.Println("No logged-in devices found. Type 'n' to scan QR for a new number.")
	}

	// Start CLI command loop.
	go cliLoop(ctx)

	// Start HTTP API server.
	go startAPIServer()

	// Start outbox scanner.
	if err := os.MkdirAll(*outboxDir, 0755); err != nil {
		log.Fatalf("Failed to create outbox directory: %v", err)
	}
	go outboxLoop(ctx)

	// Wait for signal.
	c := make(chan os.Signal, 1)
	signal.Notify(c, os.Interrupt, syscall.SIGTERM)
	<-c
	log.Println("Shutting down...")

	// Shutdown HTTP server.
	if apiServer != nil {
		sctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		apiServer.Shutdown(sctx)
	}

	for _, ci := range clients.list() {
		ci.client.Disconnect()
	}
}

// cliLoop reads stdin commands.
//
//	n / new  — scan QR for new number
//	p / pair — add new number via pair code (enter phone, get code)
//	x / logout — logout and remove a number (enter phone)
//	l / list — list connected numbers
//	q / quit — exit
func cliLoop(ctx context.Context) {
	scan := bufio.NewScanner(os.Stdin)
	fmt.Println("\nCommands: n (new/QR), p (pair code), x (logout), l (list), q (quit)")
	var pairMode bool
	var logoutMode bool
	for scan.Scan() {
		line := strings.TrimSpace(scan.Text())
		if pairMode {
			pairMode = false
			phone := strings.TrimSpace(line)
			if phone != "" {
				addNewDeviceByPair(ctx, phone)
			}
			continue
		}
		if logoutMode {
			logoutMode = false
			phone := strings.TrimSpace(line)
			if phone != "" {
				logoutDevice(phone)
			}
			continue
		}
		switch strings.ToLower(line) {
		case "n", "new":
			addNewDevice(ctx)
		case "p", "pair":
			fmt.Print("Enter phone number (e.g., 628123456789): ")
			pairMode = true
		case "x", "logout":
			fmt.Print("Enter phone number to logout (e.g., 628123456789): ")
			logoutMode = true
		case "l", "list":
			listClients()
		case "q", "quit", "exit":
			log.Println("Exit requested.")
			os.Exit(0)
		default:
			if len(line) > 0 {
				fmt.Println("Unknown command. Use: n (new), p (pair), x (logout), l (list), q (quit)")
			}
		}
	}
}

// addNewDevice creates a fresh device, shows QR, pairs it, and connects.
func addNewDevice(ctx context.Context) {
	device := storeContainer.NewDevice()

	cli := whatsmeow.NewClient(device, waLog.Stdout("Client(NEW)", logLevel, true))

	var isWaitingForPair atomic.Bool
	cli.PrePairCallback = func(jid types.JID, platform, businessName string) bool {
		isWaitingForPair.Store(true)
		defer isWaitingForPair.Store(false)
		log.Printf("Pairing %s (platform: %q, business: %q). Type 'r' within 3s to reject.", jid, platform, businessName)
		select {
		case reject := <-pairRejectChan:
			if reject {
				log.Println("Rejecting pair")
				return false
			}
		case <-time.After(3 * time.Second):
		}
		log.Println("Accepting pair")
		return true
	}

	// Channel to wait for the first successful connection (JID assigned).
	ready := make(chan struct{}, 1)
	cli.AddEventHandler(func(rawEvt interface{}) {
		if _, ok := rawEvt.(*events.Connected); ok {
			select {
			case ready <- struct{}{}:
			default:
			}
		}
	})

	ch, err := cli.GetQRChannel(ctx)
	if err != nil {
		log.Printf("Failed to get QR channel: %v", err)
		return
	}

	fmt.Println("\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
	fmt.Println("  Scan QR code below with WhatsApp")
	fmt.Println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
	go func() {
		for evt := range ch {
			switch evt.Event {
			case "code":
				qrterminal.GenerateHalfBlock(evt.Code, qrterminal.L, os.Stdout)
			case "success":
				log.Println("QR pairing successful!")
			default:
				log.Printf("QR event: %s", evt.Event)
			}
		}
	}()

	err = cli.Connect()
	if err != nil {
		log.Printf("Failed to connect new device: %v", err)
		return
	}

	// Wait for connection to be fully established (JID assigned).
	select {
	case <-ready:
	case <-time.After(60 * time.Second):
		log.Println("Timeout waiting for new device to connect")
		return
	}

	// Device is now fully connected — the JID is available.
	phone := extractPhone(device.ID)
	log.Printf("New device connected: %s", phone)

	// Check if this phone is already connected.
	if clients.hasPhone(phone) {
		log.Printf("[%s] Already connected! Disconnecting duplicate...", phone)
		cli.Disconnect()
		return
	}

	// Register handler and store in manager.
	cli.AddEventHandler(handlerFor(cli))
	clients.add(device.ID.String(), &clientInfo{client: cli, phone: phone})

	log.Printf("[%s] Ready. Total devices: %d", phone, clients.count())
}

// addNewDeviceByPair creates a new device and pairs using a phone number + code.
// User enters the code on their WhatsApp: Linked Devices → Link with phone number.
func addNewDeviceByPair(ctx context.Context, phoneNumber string) {
	// Check if this phone is already connected.
	if clients.hasPhone(phoneNumber) {
		log.Printf("[%s] Already connected! Aborting.", phoneNumber)
		return
	}

	device := storeContainer.NewDevice()
	cli := whatsmeow.NewClient(device, waLog.Stdout("Client(NEW-PAIR)", logLevel, true))

	// Channel to wait for QR event (prerequisite for PairPhone).
	qrReady := make(chan struct{}, 1)
	var qrErr error
	cli.AddEventHandler(func(rawEvt interface{}) {
		switch evt := rawEvt.(type) {
		case *events.QR:
			select {
			case qrReady <- struct{}{}:
			default:
			}
		case *events.PairSuccess:
			log.Printf("Pair code accepted! ID: %s, Business: %s", evt.ID, evt.BusinessName)
		case *events.PairError:
			log.Printf("Pair error: %v", evt.Error)
			qrErr = evt.Error
			select {
			case qrReady <- struct{}{}:
			default:
			}
		}
	})

	err := cli.Connect()
	if err != nil {
		log.Printf("Failed to connect device for pairing: %v", err)
		return
	}

	// Wait for QR event (required before calling PairPhone).
	select {
	case <-qrReady:
	case <-time.After(30 * time.Second):
		log.Println("Timeout waiting for QR readiness")
		return
	}

	if qrErr != nil {
		log.Printf("Pair error before code: %v", qrErr)
		return
	}

	// Generate the pair code.
	code, err := cli.PairPhone(ctx, phoneNumber, true, whatsmeow.PairClientChrome, "Chrome (macOS)")
	if err != nil {
		log.Printf("Failed to generate pair code: %v", err)
		return
	}

	fmt.Println("\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
	fmt.Printf("  Pair Code: %s\n", code)
	fmt.Println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")
	fmt.Println("  Open WhatsApp on your phone → Linked Devices")
	fmt.Println("  → Link with phone number → Enter the code above")
	fmt.Println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━")

	// Wait for pairing to complete (Connected or PairSuccess event).
	connected := make(chan struct{}, 1)
	cli.AddEventHandler(func(rawEvt interface{}) {
		switch rawEvt.(type) {
		case *events.Connected, *events.PairSuccess:
			select {
			case connected <- struct{}{}:
			default:
			}
		}
	})

	select {
	case <-connected:
	case <-time.After(120 * time.Second):
		log.Println("Timeout waiting for pair code to be accepted")
		return
	}

	// Device is now paired — the JID is available.
	phone := extractPhone(device.ID)
	log.Printf("New device paired via code: %s", phone)

	// Register handler and store in manager.
	cli.AddEventHandler(handlerFor(cli))
	clients.add(device.ID.String(), &clientInfo{client: cli, phone: phone})

	log.Printf("[%s] Ready. Total devices: %d", phone, clients.count())
}

// connectDevice connects a previously-paired device.
func connectDevice(ctx context.Context, device *store.Device, phone string) {
	cli := whatsmeow.NewClient(device, waLog.Stdout("Client("+phone+")", logLevel, true))

	cli.PrePairCallback = func(jid types.JID, platform, businessName string) bool {
		log.Printf("Pairing request for %s — auto-accepting (already paired)", phone)
		return true
	}

	cli.AddEventHandler(handlerFor(cli))
	err := cli.Connect()
	if err != nil {
		log.Printf("[%s] Failed to connect: %v", phone, err)
		return
	}

	clients.add(device.ID.String(), &clientInfo{client: cli, phone: phone})
	log.Printf("[%s] Connected. Total devices: %d", phone, clients.count())
}

// outboxLoop continuously scans the outbox directory for files to send.
// File format: {sender_phone}.{recipient_phone}.txt
//
//	or {sender_phone}.{recipient_phone}.{anything}.txt (extra parts ignored)
//
// Content: the message text to send.
// If sender phone not registered → random client.
// If no files or all processed → sleep 30s.
func outboxLoop(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			return
		default:
		}

		files, err := filepath.Glob(filepath.Join(*outboxDir, "*.txt"))
		if err != nil {
			log.Printf("[outbox] Error scanning directory: %v", err)
			time.Sleep(30 * time.Second)
			continue
		}

		if len(files) == 0 {
			time.Sleep(30 * time.Second)
			continue
		}

		for _, file := range files {
			processOutboxFile(ctx, file)
			// Random delay 1-10s to avoid WhatsApp rate limits.
			time.Sleep(time.Duration(1+rand.Intn(10)) * time.Second)
		}

		// Delay between scans. Longer if no clients connected.
		if clients.count() == 0 {
			log.Printf("[outbox] No clients connected, waiting 60s before retry")
			time.Sleep(60 * time.Second)
		} else {
			time.Sleep(5 * time.Second)
		}
	}
}

// processOutboxFile reads, sends, and deletes a single outbox file.
func processOutboxFile(ctx context.Context, filePath string) {
	baseName := filepath.Base(filePath)
	baseName = strings.TrimSuffix(baseName, ".txt")

	// Parse filename: {sender}.{recipient}[.{anything}].txt
	// Split by the FIRST dot only — the rest is the recipient (+ optional timestamp).
	firstDot := strings.IndexByte(baseName, '.')
	if firstDot < 0 {
		log.Printf("[outbox] Invalid filename format: %s (expected sender.recipient[.anything].txt)", baseName)
		return
	}
	senderPhone := baseName[:firstDot]
	rest := baseName[firstDot+1:]

	// Extract recipient JID from "rest".
	// Supports:
	//   628123456789                          → user JID
	//   628123456789.1718812345               → user JID (timestamp ignored)
	//   120363311558159804@g.us               → group JID
	//   120363311558159804@g.us.1718812345    → group JID (timestamp ignored)
	//   628123456789@s.whatsapp.net           → full user JID
	var recipientJID types.JID
	switch {
	case strings.HasPrefix(rest, "120363") || strings.Contains(rest, "@g.us"):
		// Group JID: extract the part up to (and including) @g.us.
		if idx := strings.Index(rest, "@g.us"); idx >= 0 {
			groupID := rest[:idx]
			recipientJID = types.NewJID(groupID, types.GroupServer)
		} else {
			// Numeric group-like ID without @g.us suffix.
			if dotIdx := strings.IndexByte(rest, '.'); dotIdx >= 0 {
				rest = rest[:dotIdx]
			}
			recipientJID = types.NewJID(rest, types.GroupServer)
		}
	case strings.Contains(rest, "@"):
		// Full JID (e.g. 628123456789@s.whatsapp.net).
		// Strip trailing timestamp if present.
		if atIdx := strings.LastIndex(rest, "@"); atIdx >= 0 {
			jidPart := rest
			if dotAfterAt := strings.IndexByte(rest[atIdx:], '.'); dotAfterAt >= 0 {
				jidPart = rest[:atIdx+dotAfterAt]
			}
			recipientJID, _ = parseJIDStr(jidPart)
		}
	default:
		// Plain phone number, optionally with timestamp.
		phone := rest
		if dotIdx := strings.IndexByte(rest, '.'); dotIdx >= 0 {
			phone = rest[:dotIdx]
		}
		recipientJID = types.NewJID(phone, types.DefaultUserServer)
	}

	if recipientJID.IsEmpty() {
		log.Printf("[outbox] Could not parse recipient JID from %s", baseName)
		return
	}

	// Read message content.
	data, err := os.ReadFile(filePath)
	if err != nil {
		log.Printf("[outbox] Error reading %s: %v", baseName, err)
		return
	}
	msg := strings.TrimSpace(string(data))
	if msg == "" {
		log.Printf("[outbox] Empty message in %s, skipping", baseName)
		os.Remove(filePath)
		return
	}

	// Find the client to send from.
	ci := clients.findByPhone(senderPhone)
	if ci == nil {
		ci = clients.random()
		if ci == nil {
			log.Printf("[outbox] No connected clients available for %s", baseName)
			return
		}
		log.Printf("[outbox] Sender %s not found, using random: %s", senderPhone, ci.phone)
	}

	if !ci.client.IsConnected() {
		log.Printf("[outbox] Client %s is not connected, skipping %s", ci.phone, baseName)
		return
	}

	// Send.
	ci.client.SendChatPresence(ctx, recipientJID, types.ChatPresence("composing"), "")
	msgr, err := ci.client.SendMessage(ctx, recipientJID, &waProto.Message{Conversation: ptr(msg)})
	if err != nil {
		log.Printf("[outbox] [%s→%s] Error sending: %v", ci.phone, recipientJID.String(), err)
		return
	}
	log.Printf("[outbox] [%s→%s] Sent (ts: %s)", ci.phone, recipientJID.String(), msgr.Timestamp)

	// Delete file after successful send.
	if err := os.Remove(filePath); err != nil {
		log.Printf("[outbox] Error deleting %s: %v", baseName, err)
	}
}

// parseJIDStr parses a full JID string like "628123456789@s.whatsapp.net".
func parseJIDStr(raw string) (types.JID, error) {
	atIdx := strings.LastIndex(raw, "@")
	if atIdx < 0 {
		return types.EmptyJID, fmt.Errorf("invalid JID: %s", raw)
	}
	user := raw[:atIdx]
	server := raw[atIdx+1:]
	return types.NewJID(user, server), nil
}

// listClients prints all connected numbers.
func listClients() {
	list := clients.list()
	if len(list) == 0 {
		fmt.Println("No connected numbers.")
		return
	}
	fmt.Println("\nConnected numbers:")
	for i, ci := range list {
		status := "disconnected"
		if ci.client.IsConnected() {
			status = "connected"
		}
		fmt.Printf("  %d. %s (%s)\n", i+1, ci.phone, status)
	}
	fmt.Println()
}

// logoutDevice disconnects and removes a device by phone number.
func logoutDevice(phone string) {
	jidKey := clients.findKeyByPhone(phone)
	if jidKey == "" {
		log.Printf("[logout] Phone %s not found in connected devices", phone)
		return
	}

	ci := clients.get(jidKey)
	if ci == nil {
		return
	}

	// Disconnect the client.
	ci.client.Disconnect()
	clients.remove(jidKey)

	// Remove from database.
	ctx := context.Background()
	jid, err := parseJIDStr(jidKey)
	if err == nil && !jid.IsEmpty() {
		device, err := storeContainer.GetDevice(ctx, jid)
		if err == nil && device != nil {
			if err := device.Delete(ctx); err != nil {
				log.Printf("[logout] Error deleting device from DB: %v", err)
			}
		}
	}

	log.Printf("[logout] %s logged out and removed. Total devices: %d", phone, clients.count())
}

// handlerFor returns an event handler that replies using the given client.
// This ensures replies always come from the correct number.
func handlerFor(cli *whatsmeow.Client) func(interface{}) {
	return func(rawEvt interface{}) {
		ctx := context.Background()
		switch evt := rawEvt.(type) {
		case *events.AppStateSyncComplete:
			if len(cli.Store.PushName) > 0 && evt.Name == appstate.WAPatchCriticalBlock {
				err := cli.SendPresence(ctx, types.PresenceAvailable)
				if err != nil {
					log.Printf("Failed to send available presence: %v", err)
				} else {
					log.Println("Marked self as available")
				}
			}
		case *events.Connected, *events.PushNameSetting:
			if len(cli.Store.PushName) == 0 {
				return
			}
			err := cli.SendPresence(ctx, types.PresenceAvailable)
			if err != nil {
				log.Printf("Failed to send available presence: %v", err)
			} else {
				log.Println("Marked self as available")
			}
		case *events.StreamReplaced:
			log.Println("Stream replaced — disconnecting this client only")
			cli.Disconnect()
		case *events.Message:
			metaParts := []string{fmt.Sprintf("pushname: %s", evt.Info.PushName), fmt.Sprintf("timestamp: %s", evt.Info.Timestamp)}
			if evt.Info.Type != "" {
				metaParts = append(metaParts, fmt.Sprintf("type: %s", evt.Info.Type))
			}
			if evt.Info.Category != "" {
				metaParts = append(metaParts, fmt.Sprintf("category: %s", evt.Info.Category))
			}
			if evt.IsViewOnce {
				metaParts = append(metaParts, "view once")
			}
			if evt.IsViewOnceV2 {
				metaParts = append(metaParts, "ephemeral (v2)")
			}
			if evt.IsDocumentWithCaption {
				metaParts = append(metaParts, "document with caption")
			}
			if evt.IsEdit {
				metaParts = append(metaParts, "edit")
			}

			// Get our own phone for logging.
			myPhone := extractPhone(cli.Store.ID)
			log.Printf("[%s] Received message %s from %s (%s)", myPhone, evt.Info.ID, evt.Info.SourceString(), strings.Join(metaParts, ", "))

			// Reply target: use Chat JID (works for DMs and groups).
			replyTarget := evt.Info.Chat

			// Real phone number: prefer SenderAlt (PN for LID senders),
			// fall back to Chat.User for regular phone number senders.
			phone := extractPhone(&evt.Info.SenderAlt)
			if phone == "unknown" || phone == "" {
				phone = extractPhone(&replyTarget)
			}
			log.Printf("[%s] Reply target: %s (phone: %s)", myPhone, replyTarget.String(), phone)

			// Extract message text.
			msg := evt.Message.GetConversation()
			if msg == "" {
				msg = evt.Message.GetExtendedTextMessage().GetText()
				if msg == "" {
					return
				}
			}

			if webhookURL != "" {
				client := resty.New()
				resp, err := client.R().
					SetFormData(map[string]string{
						"from": phone,
						"text": msg,
					}).
					Post(webhookURL)
				if err != nil {
					log.Printf("[%s] Error calling webhook: %v", myPhone, err)
					return
				}
				if respStr := resp.String(); len(respStr) > 0 && !strings.HasPrefix(strings.ToLower(respStr), "error") {
					cli.SendChatPresence(ctx, replyTarget, types.ChatPresence("composing"), "")
					time.Sleep(1 * time.Second)
					msgr, err := cli.SendMessage(ctx, replyTarget, &waProto.Message{Conversation: ptr(respStr)})
					if err != nil {
						log.Printf("[%s] Error sending reply: %v", myPhone, err)
					} else {
						log.Printf("[%s] Reply sent (server timestamp: %s)", myPhone, msgr.Timestamp)
					}
				}
			}
		}
	}
}

// extractPhone extracts the phone number portion from a JID.
// e.g. "628123456789@s.whatsapp.net" → "628123456789"
func extractPhone(jid *types.JID) string {
	if jid == nil {
		return "unknown"
	}
	s := jid.User
	// Strip gateway suffix if present, e.g. "628123456789:12" → "628123456789"
	if idx := strings.IndexByte(s, ':'); idx >= 0 {
		s = s[:idx]
	}
	return s
}

// ─── HTTP API Server ───────────────────────────────────────────────

// startAPIServer starts a lightweight HTTP API for remote management.
// Endpoints:
//
//	GET  /list        → list connected numbers (JSON)
//	POST /add/qr      → start QR pairing, returns QR code string
//	POST /add/pair    → start pair-code pairing, body: {"phone":"628xxx"}
//	POST /logout       → logout and remove device, body: {"phone":"628xxx"}
//	POST /quit        → graceful shutdown
func startAPIServer() {
	mux := http.NewServeMux()
	mux.HandleFunc("/list", apiListClients)
	mux.HandleFunc("/add/qr", apiAddQR)
	mux.HandleFunc("/add/pair", apiAddPair)
	mux.HandleFunc("/logout", apiLogout)
	mux.HandleFunc("/quit", apiQuit)

	apiServer = &http.Server{
		Addr:    apiPort,
		Handler: mux,
	}

	log.Printf("[API] Listening on %s", apiPort)
	if err := apiServer.ListenAndServe(); err != nil && err != http.ErrServerClosed {
		log.Printf("[API] Server error: %v", err)
	}
}

// apiListClients responds with JSON array of connected numbers.
// GET /list
func apiListClients(w http.ResponseWriter, r *http.Request) {
	list := clients.list()
	type item struct {
		Phone  string `json:"phone"`
		Status string `json:"status"`
	}
	result := make([]item, 0, len(list))
	for _, ci := range list {
		status := "disconnected"
		if ci.client.IsConnected() {
			status = "connected"
		}
		result = append(result, item{Phone: ci.phone, Status: status})
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"total":   len(result),
		"clients": result,
	})
}

// apiAddQR starts QR pairing and returns a PNG QR code image.
// POST /add/qr  →  Content-Type: image/png
func apiAddQR(w http.ResponseWriter, r *http.Request) {
	device := storeContainer.NewDevice()
	cli := whatsmeow.NewClient(device, waLog.Stdout("Client(API-QR)", logLevel, true))

	cli.PrePairCallback = func(jid types.JID, platform, businessName string) bool {
		log.Printf("[API-QR] Pairing %s — accepting", jid)
		return true
	}

	qrChan, err := cli.GetQRChannel(context.Background())
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]any{"error": err.Error()})
		return
	}

	err = cli.Connect()
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]any{"error": err.Error()})
		return
	}

	// Wait for first QR code (with timeout).
	select {
	case evt := <-qrChan:
		if evt.Event != "code" {
			writeJSON(w, http.StatusInternalServerError, map[string]any{"error": "QR event: " + evt.Event})
			return
		}
		// Generate PNG QR code image.
		qrCode, err := qr.Encode(evt.Code, qr.L)
		if err != nil {
			writeJSON(w, http.StatusInternalServerError, map[string]any{"error": "failed to encode QR: " + err.Error()})
			return
		}
		w.Header().Set("Content-Type", "image/png")
		w.WriteHeader(http.StatusOK)
		w.Write(qrCode.PNG())

		// Wait for connection in background and register.
		go func() {
			ready := make(chan struct{}, 1)
			cli.AddEventHandler(func(rawEvt interface{}) {
				if _, ok := rawEvt.(*events.Connected); ok {
					select {
					case ready <- struct{}{}:
					default:
					}
				}
			})
			select {
			case <-ready:
			case <-time.After(60 * time.Second):
				log.Println("[API-QR] Timeout waiting for connection")
				return
			}
			phone := extractPhone(device.ID)
			if clients.hasPhone(phone) {
				log.Printf("[API-QR] %s already connected, disconnecting duplicate", phone)
				cli.Disconnect()
				return
			}
			cli.AddEventHandler(handlerFor(cli))
			clients.add(device.ID.String(), &clientInfo{client: cli, phone: phone})
			log.Printf("[API-QR] %s connected. Total: %d", phone, clients.count())
		}()
	case <-time.After(30 * time.Second):
		writeJSON(w, http.StatusInternalServerError, map[string]any{"error": "timeout waiting for QR code"})
	}
}

// apiAddPair starts pair-code pairing. Body: {"phone": "628123456789"}
// POST /add/pair
func apiAddPair(w http.ResponseWriter, r *http.Request) {
	var body struct {
		Phone string `json:"phone"`
	}
	if err := json.NewDecoder(r.Body).Decode(&body); err != nil || body.Phone == "" {
		writeJSON(w, http.StatusBadRequest, map[string]any{"error": "missing or invalid 'phone' in JSON body"})
		return
	}
	if clients.hasPhone(body.Phone) {
		writeJSON(w, http.StatusConflict, map[string]any{"error": "phone already connected"})
		return
	}

	device := storeContainer.NewDevice()
	cli := whatsmeow.NewClient(device, waLog.Stdout("Client(API-PAIR)", logLevel, true))

	qrReady := make(chan struct{}, 1)
	var qrErr error
	cli.AddEventHandler(func(rawEvt interface{}) {
		switch evt := rawEvt.(type) {
		case *events.QR:
			select {
			case qrReady <- struct{}{}:
			default:
			}
		case *events.PairError:
			qrErr = evt.Error
			select {
			case qrReady <- struct{}{}:
			default:
			}
		}
	})

	err := cli.Connect()
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]any{"error": err.Error()})
		return
	}

	select {
	case <-qrReady:
	case <-time.After(30 * time.Second):
		writeJSON(w, http.StatusInternalServerError, map[string]any{"error": "timeout waiting for QR readiness"})
		return
	}

	if qrErr != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]any{"error": qrErr.Error()})
		return
	}

	code, err := cli.PairPhone(context.Background(), body.Phone, true, whatsmeow.PairClientChrome, "Chrome (macOS)")
	if err != nil {
		writeJSON(w, http.StatusInternalServerError, map[string]any{"error": err.Error()})
		return
	}

	writeJSON(w, http.StatusOK, map[string]any{
		"pair_code": code,
		"message":   "Enter this code on WhatsApp → Linked Devices → Link with phone number. Check /list for status.",
	})

	// Wait for connection in background and register.
	go func() {
		connected := make(chan struct{}, 1)
		cli.AddEventHandler(func(rawEvt interface{}) {
			switch rawEvt.(type) {
			case *events.Connected, *events.PairSuccess:
				select {
				case connected <- struct{}{}:
				default:
				}
			}
		})
		select {
		case <-connected:
		case <-time.After(120 * time.Second):
			log.Println("[API-PAIR] Timeout waiting for pair code acceptance")
			return
		}
		phone := extractPhone(device.ID)
		if clients.hasPhone(phone) {
			log.Printf("[API-PAIR] %s already connected, disconnecting duplicate", phone)
			cli.Disconnect()
			return
		}
		cli.AddEventHandler(handlerFor(cli))
		clients.add(device.ID.String(), &clientInfo{client: cli, phone: phone})
		log.Printf("[API-PAIR] %s connected. Total: %d", phone, clients.count())
	}()
}

// apiLogout logs out and removes a device. Body: {"phone": "628123456789"}
// POST /logout
func apiLogout(w http.ResponseWriter, r *http.Request) {
	var body struct {
		Phone string `json:"phone"`
	}
	if err := json.NewDecoder(r.Body).Decode(&body); err != nil || body.Phone == "" {
		writeJSON(w, http.StatusBadRequest, map[string]any{"error": "missing or invalid 'phone' in JSON body"})
		return
	}
	if !clients.hasPhone(body.Phone) {
		writeJSON(w, http.StatusNotFound, map[string]any{"error": "phone not found in connected devices"})
		return
	}
	logoutDevice(body.Phone)
	writeJSON(w, http.StatusOK, map[string]any{"message": fmt.Sprintf("%s logged out", body.Phone)})
}

// apiQuit triggers graceful shutdown.
// POST /quit
func apiQuit(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{"message": "shutting down"})
	go func() {
		time.Sleep(100 * time.Millisecond)
		// Send SIGTERM to self.
		p, _ := os.FindProcess(os.Getpid())
		p.Signal(syscall.SIGTERM)
	}()
}

func writeJSON(w http.ResponseWriter, status int, data any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(data)
}
